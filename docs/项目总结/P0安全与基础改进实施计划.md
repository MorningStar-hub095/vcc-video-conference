**改进目标：** 创建一个 bcrypt 密码哈希封装模块，利用 Linux 系统 `crypt()` 函数，提供 `bcrypt_hash(password)` 和 `bcrypt_verify(password, hash)` 两个接口。

- [ ] **Step 1: 创建 bcrypt_wrapper.h 头文件**

```cpp
#ifndef BCRYPT_WRAPPER_H
#define BCRYPT_WRAPPER_H

#include <string>

// [P0新增] bcrypt 密码哈希封装模块
// 基于 Linux 系统 crypt() 函数实现 bcrypt 哈希
// 替代原有 MD5 密码存储方案，大幅提升抗彩虹表/暴力破解能力
// 依赖：Linux libcrypt（链接时需 -lcrypt）

// 生成 bcrypt 哈希
// 参数：password - 待哈希的密码（客户端发来的 MD5 哈希值）
// 返回：bcrypt 哈希字符串（格式：$2b$12$<22字节盐><31字节哈希>）
// 失败返回空字符串
std::string bcrypt_hash(const std::string& password);

// 验证密码是否匹配 bcrypt 哈希
// 参数：password - 待验证的密码（客户端发来的 MD5 哈希值）
//       hash - 数据库存储的 bcrypt 哈希字符串
// 返回：true=匹配成功，false=不匹配或错误
bool bcrypt_verify(const std::string& password, const std::string& hash);

#endif // BCRYPT_WRAPPER_H
```

- [ ] **Step 2: 创建 bcrypt_wrapper.cpp 实现文件**

创建文件 `server\PlayerHallServer\src\bcrypt_wrapper.cpp`：

```cpp
#define _GNU_SOURCE  // [P0新增] 启用 GNU 扩展，使 crypt() 可用
#include "bcrypt_wrapper.h"

#include <crypt.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

// [P0新增] 生成随机 bcrypt 盐值
// bcrypt 盐值格式：$2b$<cost>$<22字节base64盐>
// cost=12 表示 2^12=4096 轮迭代，平衡安全性和性能
static std::string generate_bcrypt_salt()
{
    // bcrypt 算法标识 + cost 参数
    const char prefix[] = "$2b$12$";
    // 22 字节 base64 盐值字符集（bcrypt 使用的变体 base64）
    const char b64_chars[] = "./ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    
    // 使用随机数生成盐值
    // 注意：生产环境应使用 /dev/urandom，此处用 rand()+time() 作为简化方案
    unsigned int seed = (unsigned int)time(NULL) ^ (unsigned int)rand();
    srand(seed);
    
    char salt[32] = {0};
    strncpy(salt, prefix, sizeof(salt) - 1);
    for (int i = 7; i < 29 && i < (int)sizeof(salt) - 1; i++) {
        salt[i] = b64_chars[rand() % 64];
    }
    salt[29] = '\0';
    
    return std::string(salt);
}

std::string bcrypt_hash(const std::string& password)
{
    std::string salt = generate_bcrypt_salt();
    
    // 调用系统 crypt() 函数进行 bcrypt 哈希
    // crypt() 线程不安全，需用 crypt_r() 线程安全版本
    struct crypt_data data;
    memset(&data, 0, sizeof(data));
    
    char* result = crypt_r(password.c_str(), salt.c_str(), &data);
    if (result == NULL || result[0] == '\0') {
        printf("[bcrypt] hash failed, password len=%zu\n", password.size());
        return "";
    }
    
    return std::string(result);
}

bool bcrypt_verify(const std::string& password, const std::string& hash)
{
    if (hash.empty() || hash.size() < 7) {
        return false;
    }
    
    // 使用存储的哈希值作为盐值（bcrypt 哈希值本身包含盐值）
    // crypt() 会从 hash 字符串中解析出算法、cost、盐值
    struct crypt_data data;
    memset(&data, 0, sizeof(data));
    
    char* result = crypt_r(password.c_str(), hash.c_str(), &data);
    if (result == NULL) {
        return false;
    }
    
    // 比较生成的哈希与存储的哈希是否一致
    // 使用常量时间比较防止时序攻击
    int rc = strcmp(result, hash.c_str());
    return rc == 0;
}
```

- [ ] **Step 3: 处理数据库中旧密码的兼容（数据迁移说明）**

> **重要：** 数据库中已注册用户的密码仍是 MD5 哈希值（非 bcrypt 格式）。`bcrypt_verify` 会检测到哈希值不以 `$2b$` 开头，返回 false，导致老用户无法登录。

**兼容方案：** 在 LoginRq 中添加自动迁移逻辑——如果数据库密码不以 `$2b$` 开头（旧 MD5 格式），先用 strcmp 验证，验证成功后自动升级为 bcrypt：

```cpp
// [P0新增] 密码格式兼容与自动迁移
// 旧密码(MD5)不以$2b$开头，新密码(bcrypt)以$2b$开头
bool bPwdCorrect = false;
if (strPwd.size() >= 4 && strPwd.substr(0, 4) == "$2b$") {
    // 新格式：bcrypt 验证
    bPwdCorrect = bcrypt_verify(rq->m_password, strPwd);
} else {
    // 旧格式：MD5 直接比较 + 自动迁移
    bPwdCorrect = (strcmp(rq->m_password, strPwd.c_str()) == 0);
    if (bPwdCorrect) {
        // 自动迁移：将旧 MD5 哈希值升级为 bcrypt 哈希
        std::string strNewPwd = bcrypt_hash(rq->m_password);
        if (!strNewPwd.empty()) {
            char strUpdate[512] = "";
            sprintf(strUpdate, "update t_user set password='%s' where id=%s;",
                    strNewPwd.c_str(), strId.c_str());
            m_sql->UpdataMysql(strUpdate);
            printf("[LoginRq] 用户 %s 密码已自动迁移到 bcrypt\n", strId.c_str());
        }
    }
}

if (!bPwdCorrect) {
    rs.m_lresult = password_error;
} else {
    // 密码正确，继续原有逻辑...
}
```

// [P0新增] AEC 回声消除处理器
// 基于 Speex DSP 的回声消除模块
// 工作原理：
//   1. 播放端调用 playback() 提供远端参考信号（扬声器即将播放的声音）
//   2. 采集端调用 capture() 传入麦克风数据，AEC 消除其中的回声成分

    // sampleRate: 采样率（48000）
    // filterLength: 滤波器长度（采样数，建议为 0.25 秒 = 12000 @ 48kHz）
    bool init(int frameSize = 960, int sampleRate = 48000, int filterLength = 12000);

    void setEnabled(bool enabled) { m_enabled = enabled; }

    int m_frameSize;                 // 帧大小（采样数）

    std::mutex m_mutex;
    int16_t m_farBuf[960];           // far-end 参考缓冲（一帧大小）

#include <QDebug>

AecProcessor::AecProcessor(QObject *parent)
    : QObject(parent)
    , m_echoState(nullptr)
    , m_preprocessState(nullptr)
    , m_frameSize(960)

AecProcessor::~AecProcessor()
{
    if (m_preprocessState) {
        speex_preprocess_state_destroy(m_preprocessState);
        m_preprocessState = nullptr;
    }
    if (m_echoState) {
        speex_echo_state_destroy(m_echoState);
        m_echoState = nullptr;
    }
    m_initialized = false;
}

bool AecProcessor::init(int frameSize, int sampleRate, int filterLength)
{
    m_frameSize = frameSize;

    // 创建 Speex AEC 状态
    m_echoState = speex_echo_state_init(frameSize, filterLength);
    if (!m_echoState) {
        qWarning() << "[AEC] speex_echo_state_init failed";
        return false;
    }
    speex_echo_ctl(m_echoState, SPEEX_ECHO_SET_SAMPLING_RATE, &sampleRate);

    // 创建预处理状态（降噪 + AGC，配合 AEC 使用效果更好）
    m_preprocessState = speex_preprocess_state_init(frameSize, sampleRate);
    if (!m_preprocessState) {
        qWarning() << "[AEC] speex_preprocess_state_init failed";
        speex_echo_state_destroy(m_echoState);
        m_echoState = nullptr;
        return false;
    }

    // 将 AEC 状态关联到预处理（预处理会使用 AEC 的残留回声估计）
    speex_preprocess_ctl(m_preprocessState, SPEEX_PREPROCESS_SET_ECHO_STATE, m_echoState);

    // 启用降噪（默认 15dB 降噪）
    int denoise = 1;
    int noiseSuppress = -30;
    speex_preprocess_ctl(m_preprocessState, SPEEX_PREPROCESS_SET_DENOISE, &denoise);
    speex_preprocess_ctl(m_preprocessState, SPEEX_PREPROCESS_SET_NOISE_SUPPRESS, &noiseSuppress);

    // 启用 AGC（自动增益控制）
    int agc = 1;
    speex_preprocess_ctl(m_preprocessState, SPEEX_PREPROCESS_SET_AGC, &agc);

    m_initialized = true;
    m_enabled = true;
    qDebug() << "[AEC] 初始化成功, frameSize=" << frameSize
             << "sampleRate=" << sampleRate << "filterLength=" << filterLength;
    return true;
}

void AecProcessor::playback(const int16_t* pcmData)
{
    if (!m_initialized || !m_enabled) return;
    if (!pcmData) return;

    // 将播放数据作为 far-end 参考信号送入 AEC
    speex_echo_playback(m_echoState, (const spx_int16_t*)pcmData);
}

void AecProcessor::capture(const int16_t* input, int16_t* output)
{
    if (!m_initialized || !m_enabled) {
        // AEC 未启用，直接拷贝
        if (input && output) {
            memcpy(output, input, m_frameSize * sizeof(int16_t));
        }
        return;
    }

    // AEC 回声消除
    speex_echo_capture(m_echoState, (const spx_int16_t*)input, (spx_int16_t*)output);

    // 预处理（降噪 + AGC）
    speex_preprocess_run(m_preprocessState, (spx_int16_t*)output);
}

## 风险评估

| 风险项 | 概率 | 影响 | 缓解措施 |
|--------|------|------|----------|
| **bcrypt 旧密码兼容问题** | 中 | 高 | Task 3 的自动迁移逻辑：检测旧 MD5 格式，登录成功后自动升级 |
| **AEC far-end 时序不对齐** | 中 | 中 | Speex AEC 内部有缓冲队列，播放和采集无需严格对齐；若效果差可调大 filterLength |
| **AEC 导致音频失真** | 低 | 中 | 可通过 `setEnabled(false)` 快速禁用；Speex AEC 默认参数经过广泛验证 |
| **会议密码协议不兼容** | 中 | 高 | 客户端和服务端必须同步修改 packdef.h；建议同一次部署 |
| **服务端 -lcrypt 链接失败** | 低 | 高 | 确认 Linux 系统已安装 libcrypt-dev（`apt install libcrypt-dev`） |
| **Speex DSP 库链接失败（客户端）** | 低 | 中 | 项目已有 `libspeexdsp.lib`，路径在 `AudioApi/speex/lib/` |
