```cpp
//实现：
//  - 若当前未最大化(m_isMaximized == false)：
//      1. 保存当前窗口几何信息到 m_normalGeometry（用于后续还原）
//      2. 获取当前屏幕的可用几何区域（排除任务栏）
//      3. 将窗口设置为该屏幕的可用区域大小，实现最大化效果
//      4. 标记 m_isMaximized = true
//  - 若当前已最大化(m_isMaximized == true)：
//      1. 从 m_normalGeometry 恢复窗口原始位置和大小
//      2. 标记 m_isMaximized = false
void RoomDialog::on_pb_max_clicked()
{
    if( !m_isMaximized )
    {
        //保存当前窗口几何信息，用于还原
        m_normalGeometry = this->geometry();
        //获取窗口所在屏幕的可用几何区域（排除任务栏）
        QScreen * screen = QApplication::primaryScreen();
        if( screen )
        {
            QRect deskRect = screen->availableGeometry();
            this->setGeometry( deskRect );
        }
        m_isMaximized = true;
    }else{
        //还原窗口到原始位置和大小
        this->setGeometry( m_normalGeometry );
        m_isMaximized = false;
    }
}
```

```cpp
#include <QRegExp>
#include <QMessageBox>
//提交登录信息
void LoginDialog::on_pb_commit_clicked()
{
    QString strTel      = ui->le_tel->text();
    QString strPassword = ui->le_password->text();
    //校验
    //非空校验 清掉空格后，不能是空字符串
    QString tmpTel = strTel, tmpPass = strPassword;
    if(tmpTel.remove(" ").isEmpty() || tmpPass.remove(" ").isEmpty())
    {
        QMessageBox::about(this, "提示", "手机号密码不能为空格");
        return;
    }
    //【拓展优化-修改】登录校验逻辑支持手机号/用户名/邮箱三种方式
    //原代码仅校验手机号格式：^1[3-9][0-9]{9}$
    //修改后：
    //  - 包含 '@' → 邮箱格式校验
    //  - 全数字且11位 → 手机号格式校验
    //  - 其他 → 用户名校验（非空且长度<=10）
    bool bValid = false;
    if( strTel.contains('@') )
    {
        //邮箱登录：简单校验邮箱格式
        QRegExp regEmail("^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$");
        bValid = regEmail.exactMatch(strTel);
        if(!bValid)
        {
            QMessageBox::about(this, "提示", "邮箱格式错误");
            return;
        }
    }
    else if( strTel.length() == 11 && !strTel.contains(QRegExp("[^0-9]")) )
    {
        //手机号登录：校验手机号格式
        QRegExp reg("^1[3-9][0-9]{9}$");
        bValid = reg.exactMatch(strTel);
        if(!bValid)
        {
            QMessageBox::about(this, "提示", "手机号格式错误，8-11位手机号");
            return;
        }
    }
    else
    {
        //用户名（昵称）登录：校验长度
        if( strTel.length() > 10 || strTel.length() < 1 )
        {
            QMessageBox::about(this, "提示", "用户名长度需在1-10之间");
            return;
        }
        bValid = true;
```

```cpp
//【拓展优化-新增】预定会议按钮点击槽函数
//功能：预定一个未来的会议，生成会议号和会议时间，保存到本地配置文件
//实现：
//  1. 弹出输入对话框让用户输入会议主题
//  2. 弹出输入对话框让用户选择会议时长（分钟）
//  3. 生成唯一的会议号（基于当前时间戳）
//  4. 将会议信息（会议号、主题、时间、时长）保存到本地配置文件 booking.ini
//  5. 显示预定成功提示，包含会议号和会议时间
void WeChatDialog::on_toolButton_3_clicked()
{
    //1.输入会议主题
    bool ok = false;
    QString topic = QInputDialog::getText(this, tr("预定会议"),
                                          tr("请输入会议主题:"), QLineEdit::Normal,
                                          tr("我的会议"), &ok);
    if( !ok || topic.isEmpty() )
    {
        return;
    }

    //2.输入会议时长（分钟）
    QStringList items;
    items << tr("30分钟") << tr("60分钟") << tr("90分钟") << tr("120分钟");
    QString item = QInputDialog::getItem(this, tr("预定会议"),
                                          tr("请选择会议时长:"), items, 0, false, &ok);
    if( !ok || item.isEmpty() )
    {
        return;
    }

    //解析时长
    int duration = 30;
    if( item.startsWith("60") ) duration = 60;
    else if( item.startsWith("90") ) duration = 90;
    else if( item.startsWith("120") ) duration = 120;

    //3.生成会议号（基于当前时间戳，确保唯一性）
    //格式：MMDDHHMM（月日时分）
    qint64 timestamp = QDateTime::currentDateTime().toSecsSinceEpoch();
    int roomid = (int)(timestamp % 100000000);  //取后8位

    //4.计算会议开始时间（当前时间 + 5分钟提醒缓冲）
    QDateTime startTime = QDateTime::currentDateTime().addSecs(5 * 60);
    QString strStartTime = startTime.toString("yyyy-MM-dd HH:mm");

    //5.保存到本地配置文件 booking.ini
    //配置文件保存在程序所在目录下，方便用户查看
    QString configPath = QCoreApplication::applicationDirPath() + "/booking.ini";
    QSettings settings(configPath, QSettings::IniFormat);
    settings.beginGroup("Booking_" + QString::number(roomid));
    settings.setValue("roomid", roomid);
    settings.setValue("topic", topic);
    settings.setValue("starttime", strStartTime);
    settings.setValue("duration", duration);
    settings.endGroup();

    //6.显示预定成功提示
    QString info = tr("会议预定成功！\n\n")
                 + tr("会议号: %1\n").arg(roomid)
                 + tr("会议主题: %1\n").arg(topic)
                 + tr("开始时间: %1\n").arg(strStartTime)
                 + tr("会议时长: %1分钟\n").arg(duration)
                 + tr("\n请记下会议号，到时间后使用\"加入会议\"功能输入会议号加入。");
    QMessageBox::information(this, tr("预定成功"), info);

    //7.发送信号通知 ckernel（预留扩展，可用于服务端同步）
    Q_EMIT SIG_bookRoom();
}
```

#### 修改4：新增 UpdateEmailRq 函数

##### 新增代码

```cpp
//【拓展优化-新增】补全邮箱请求处理
//功能：已注册用户登录后若邮箱为空，客户端弹出对话框让用户补全邮箱
//用户提交后通过此函数更新数据库中的 email 字段
//流程：
//  1. 拆包获取 userid 和 email
//  2. 校验邮箱格式（简单校验：包含 @ 和 .）
//  3. 检查邮箱是否已被其他用户使用
//  4. 更新 t_user 表中该 userid 的 email 字段
//  5. 返回补全结果
void CLogic::UpdateEmailRq(int clientfd, char * szbuf, int nlen)
{
    printf("clientfd:%d UpdateEmailRq\n", clientfd);
    //1.拆包
    STRU_UPDATE_EMAIL_RQ * rq = (STRU_UPDATE_EMAIL_RQ *)szbuf;
    STRU_UPDATE_EMAIL_RS rs;

    //2.简单校验邮箱格式（包含 @ 和 .）
    string strEmail = rq->m_email;
    if( strEmail.find('@') == string::npos || strEmail.find('.') == string::npos )
    {
        rs.m_lresult = 0; //失败：邮箱格式错误
        SendData(clientfd, (char*)&rs, sizeof(rs));
        return;
    }

    //3.检查邮箱是否已被其他用户使用
    char sqlStr[1024] = {0};
    list<string> resList;
    sprintf(sqlStr, "select id from t_user where email = '%s' and id != %d;", rq->m_email, rq->m_userid);
    if(!m_sql->SelectMysql(sqlStr, 1, resList))
    {
        printf("SelectMysql error: %s \n", sqlStr);
        rs.m_lresult = 0;
        SendData(clientfd, (char*)&rs, sizeof(rs));
        return;
    }
    if(resList.size() > 0)
    {
        //邮箱已被其他用户使用
        rs.m_lresult = 0;
        SendData(clientfd, (char*)&rs, sizeof(rs));
        return;
    }

    //4.更新 t_user 表中该 userid 的 email 字段
    sprintf(sqlStr, "update t_user set email = '%s' where id = %d;", rq->m_email, rq->m_userid);
    if(!m_sql->UpdataMysql(sqlStr))
    {
        printf("UpdataMysql error: %s \n", sqlStr);
        rs.m_lresult = 0;
        SendData(clientfd, (char*)&rs, sizeof(rs));
        return;
    }

    //5.返回补全成功
    rs.m_lresult = 1;
    SendData(clientfd, (char*)&rs, sizeof(rs));
}
```

## 拓展优化五：登录方式改为下拉框选择

### 问题原因

##### 原代码

```cpp
//【拓展优化-修改】登录校验逻辑支持手机号/用户名/邮箱三种方式
//原代码仅校验手机号格式：^1[3-9][0-9]{9}$
//修改后：
//  - 包含 '@' → 邮箱格式校验
//  - 全数字且11位 → 手机号格式校验
//  - 其他 → 用户名校验（非空且长度<=10）
bool bValid = false;
if( strTel.contains('@') )
{
    //邮箱登录：简单校验邮箱格式
    QRegExp regEmail("^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$");
    bValid = regEmail.exactMatch(strTel);
    if(!bValid)
    {
        QMessageBox::about(this, "提示", "邮箱格式错误");
        return;
    }
}
else if( strTel.length() == 11 && !strTel.contains(QRegExp("[^0-9]")) )
{
    //手机号登录：校验手机号格式
    QRegExp reg("^1[3-9][0-9]{9}$");
    bValid = reg.exactMatch(strTel);
    if(!bValid)
    {
        QMessageBox::about(this, "提示", "手机号格式错误，8-11位手机号");
        return;
    }
}
else
{
    //用户名（昵称）登录：校验长度
    if( strTel.length() > 10 || strTel.length() < 1 )
    {
        QMessageBox::about(this, "提示", "用户名长度需在1-10之间");
        return;
    }
    bValid = true;
}
```

##### 修改后代码

```cpp
//【拓展优化-修改】根据下拉框选择的登录方式进行不同的校验

//修改后：用户通过 cb_login_type 下拉框选择登录方式

}
else if( loginType == 1 )

        return;

else if( loginType == 2 )
```

```cpp
//【拓展优化-新增】登录方式下拉框切换槽函数

//  - 索引2：邮箱 → lb_tel 显示"邮箱"
void LoginDialog::on_cb_login_type_currentIndexChanged(int index)
{
    switch(index)
    {
    case 0:
        ui->lb_tel->setText("手机号");
        break;
    case 1:
        ui->lb_tel->setText("用户名");
        break;
    case 2:
        ui->lb_tel->setText("邮箱");
        break;
    default:
        break;
    }
}
```

## 拓展优化六：修复最大化后布局问题

### 问题原因
`roomdialog.ui` 中 `wdg_bottom` 控件（包含音频/视频/桌面共享复选框和退出房间按钮）使用绝对定位（geometry y=610）放置在 `wdg_UserShow` 内部。当窗口最大化时，`wdg_UserShow` 尺寸变大，但 `wdg_bottom` 仍然固定在 y=610 位置，不会自动跟随到底部，导致布局错乱。

### 涉及文件
- 客户端：`WeChat/roomdialog.ui`

```xml
      导致窗口最大化时 wdg_bottom 不会自动跟随到底部
      修改后：改为 QVBoxLayout(verticalLayout_left)，wdg_UserShow 设置 sizePolicy=Expanding 自动拉伸
```

### 修改要点

1. **wdg_left 布局类型**：从 `QHBoxLayout`（`horizontalLayout_4`）改为 `QVBoxLayout`（`verticalLayout_left`），使子控件垂直排列
2. **wdg_UserShow 设置 sizePolicy**：设置为 `Expanding/Expanding`，当窗口最大化时自动等比例拉伸填充空间
3. **wdg_bottom 移出 wdg_UserShow**：从 `wdg_UserShow` 的子控件（绝对定位）改为 `wdg_left` 布局的第二个 item，通过垂直布局自动放置在 `wdg_UserShow` 下方
4. **wdg_bottom 去掉 geometry**：不再使用绝对定位的 `geometry` 属性，改为通过布局管理器自动定位
5. **wdg_bottom minimumSize width 改为 0**：原为 800（固定宽度），改为 0 允许自适应宽度

### 修复效果

- 窗口正常大小时：`wdg_UserShow` 在上，`wdg_bottom` 在下，布局与原来一致
- 窗口最大化时：`wdg_UserShow` 自动等比例拉伸填充空间，`wdg_bottom` 自动跟随到底部，所有控件保持正确布局
